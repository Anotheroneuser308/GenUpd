import { useState, useEffect } from 'react';
import { useGameLoop } from './hooks/useGameLoop';
import { Header } from './components/Header';
import { PointsDisplay } from './components/PointsDisplay';
import { GeneratorsTab } from './components/GeneratorsTab';
import { PrestigeTab } from './components/PrestigeTab';
import { InfinityTab } from './components/InfinityTab';
import { InfinityOverlay } from './components/InfinityOverlay';
import { AchievementsTab } from './components/AchievementsTab';
import { StatsTab } from './components/StatsTab';
import { ToastContainer } from './components/ToastContainer';

export default function App() {
  const {
    gameState,
    offlineMessage,
    showInfinityOverlay,
    proceedInfinity,
    clearOfflineMessage,
    toasts,
    dismissToast,
    buyGenerator,
    buyMaxGenerator,
    buyTickspeed,
    buyMaxAll,
    performPrestige,
    performPrestigeEnergy,
    buyPrestigeUpgrade,
    buyInfinityTreeNode,
    setAutobuyerTarget,
    setPrestigeAutobuyerSettings,
    restoreBackup,
    toggleAutobuyer,
    toggleSound,
    setNotation,
    saveGame,
    setAutoSaveEnabled,
    setAutoSaveInterval,
    exportSave,
    importSave,
    exportSaveToFile,
    importSaveFromFile,
    hardReset,
  } = useGameLoop();

  const [activeTab, setActiveTab] = useState<
    'generators' | 'prestige' | 'infinity' | 'achievements' | 'stats'
  >('generators');

  const handleProceedInfinity = () => {
    proceedInfinity();
  };

  // Keyboard Shortcuts (supporting both Latin and Cyrillic keyboard layouts)
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Ignore when focused in text input
      if (
        e.target instanceof HTMLInputElement ||
        e.target instanceof HTMLTextAreaElement
      ) {
        return;
      }

      const key = e.key.toLowerCase();

      if (key === 'm' || key === 'ь') {
        e.preventDefault();
        buyMaxAll();
      } else if (key === 'p' || key === 'з') {
        e.preventDefault();
        performPrestige();
      } else if (key === 't' || key === 'е') {
        e.preventDefault();
        buyTickspeed();
      } else if (key >= '1' && key <= '8') {
        const num = parseInt(key, 10);
        buyGenerator(num);
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [
    buyMaxAll,
    performPrestige,
    buyTickspeed,
    buyGenerator,
  ]);

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans selection:bg-indigo-500 selection:text-white relative">
      {/* Toast Notification Container */}
      <ToastContainer toasts={toasts} onDismiss={dismissToast} />

      {/* Dramatic Orange Fullscreen Infinity Overlay */}
      {showInfinityOverlay && (
        <InfinityOverlay onProceed={handleProceedInfinity} />
      )}

      {/* Top Bar Contract Header */}
      <Header
        state={gameState}
        activeTab={activeTab}
        onTabChange={setActiveTab}
        onBuyMaxAll={buyMaxAll}
        onSaveGame={saveGame}
      />

      {/* Hero Points & Status Bar */}
      <PointsDisplay state={gameState} />

      {/* Offline Catch-up Notification Banner */}
      {offlineMessage && (
        <div className="bg-indigo-950/80 border-b border-indigo-500/30 px-6 py-3 flex items-center justify-between text-xs text-indigo-200">
          <div className="flex items-center gap-2">
            <span className="font-semibold text-white">⚡</span>
            <span>{offlineMessage}</span>
          </div>
          <button
            type="button"
            onClick={clearOfflineMessage}
            className="px-2 py-1 bg-indigo-900/60 hover:bg-indigo-800 text-indigo-100 rounded text-[11px] font-medium transition-colors cursor-pointer"
          >
            Dismiss
          </button>
        </div>
      )}

      {/* Main Content Area */}
      <main className="flex-1 max-w-7xl w-full mx-auto p-6 md:p-8">
        {activeTab === 'generators' && (
          <GeneratorsTab
            state={gameState}
            onBuyGenerator={buyGenerator}
            onBuyMaxGenerator={buyMaxGenerator}
            onBuyTickspeed={buyTickspeed}
          />
        )}

        {activeTab === 'prestige' && (
          <PrestigeTab
            state={gameState}
            onPrestige={performPrestige}
            onPrestigeEnergy={performPrestigeEnergy}
            onBuyUpgrade={buyPrestigeUpgrade}
          />
        )}

        {activeTab === 'infinity' && (gameState.infinity.unlocked || gameState.infinity.count > 0 || gameState.stats.totalInfinities > 0) && (
          <InfinityTab
            state={gameState}
            onBuyInfinityTreeNode={buyInfinityTreeNode}
            onToggleAutobuyer={toggleAutobuyer}
            onSetAutobuyerTarget={setAutobuyerTarget}
            onSetPrestigeAutobuyerSettings={setPrestigeAutobuyerSettings}
          />
        )}

        {activeTab === 'achievements' && (
          <AchievementsTab state={gameState} />
        )}

        {activeTab === 'stats' && (
          <StatsTab
            state={gameState}
            onSetNotation={setNotation}
            onToggleSound={toggleSound}
            onSaveGame={saveGame}
            onSetAutoSaveEnabled={setAutoSaveEnabled}
            onSetAutoSaveInterval={setAutoSaveInterval}
            onExportSave={exportSave}
            onImportSave={importSave}
            onExportSaveToFile={exportSaveToFile}
            onImportSaveFromFile={importSaveFromFile}
            onRestoreBackup={restoreBackup}
            onHardReset={hardReset}
          />
        )}
      </main>

      {/* Clean Quiet Footer */}
      <footer className="border-t border-slate-900 px-6 py-4 text-xs text-slate-500">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-3 font-mono">
          <div className="flex items-center gap-2 text-[11px]">
            <span>Hotkeys:</span>
            <span className="text-slate-400">1-8 (Gens)</span>
            <span>·</span>
            <span className="text-slate-400">M (Max All)</span>
            <span>·</span>
            <span className="text-slate-400">T (Tickspeed)</span>
            <span>·</span>
            <span className="text-slate-400">P (Prestige)</span>
          </div>

          <div className="text-[11px] text-slate-600">
            {gameState.settings.autoSaveEnabled
              ? `Auto-saves locally every ${gameState.settings.autoSaveInterval}s`
              : 'Auto-save disabled'}
          </div>

      </footer>
    </div>
  );

import React, { useState } from 'react';

  onSetPrestigeAutobuyerSettings: (settings: Partial<{ target: number; mode: PrestigeAutobuyerMode; timeTarget: number; ratioTarget: number }>) => void;
}

export const AutomationTab: React.FC<AutomationTabProps> = ({
  state,
  onToggleAutobuyer,
  onSetAutobuyerTarget,
  onSetPrestigeAutobuyerSettings,
}) => {

  const tree = state.infinityTree || {};
  const isHyperSpeed = !!tree['auto_4'];

  const prestigeBuyer = state.autobuyers?.prestige;
  const prestigeMode: PrestigeAutobuyerMode = prestigeBuyer?.mode || 'targetPP';
  const [prestigeInput, setPrestigeInput] = useState(() => String(prestigeBuyer?.target ?? 1));
  const [timeInput, setTimeInput] = useState(() => String(prestigeBuyer?.timeTarget ?? 60));
  const [ratioInput, setRatioInput] = useState(() => String(prestigeBuyer?.ratioTarget ?? 2));
  const commitNumber = (field: 'target' | 'timeTarget' | 'ratioTarget') => {
    const raw = field === 'target' ? prestigeInput : field === 'timeTarget' ? timeInput : ratioInput;
    const value = Number(raw);
    if (!Number.isFinite(value) || value <= 0) {
      if (field === 'target') setPrestigeInput(String(prestigeBuyer?.target ?? 1));
      else if (field === 'timeTarget') setTimeInput(String(prestigeBuyer?.timeTarget ?? 60));
      else setRatioInput(String(prestigeBuyer?.ratioTarget ?? 2));
      return;
    }
    if (field === 'target') {
      const next = Math.max(1, value);
      onSetAutobuyerTarget('prestige', next);
      setPrestigeInput(String(next));
    } else if (field === 'timeTarget') {
      const next = Math.max(1, value);
      onSetPrestigeAutobuyerSettings({ timeTarget: next });
      setTimeInput(String(next));
    } else {
      const next = Math.max(1.01, value);
      onSetPrestigeAutobuyerSettings({ ratioTarget: next });
      setRatioInput(String(next));
    }
  };

  const handleToggleAll = (enable: boolean) => {
    Object.keys(state.autobuyers).forEach((key) => {
      const ab = state.autobuyers[key];
      if (ab?.unlocked && ab.enabled !== enable) {
        onToggleAutobuyer(key);
      }
    });

    { key: 'gen3', name: 'Auto: Generator 3', desc: 'Automatically purchases Generator 3', reqNode: 'Core Autobuyers (auto_1)' },
    { key: 'gen4', name: 'Auto: Generator 4', desc: 'Automatically purchases Generator 4', reqNode: 'Core Autobuyers (auto_1)' },
    { key: 'gen5', name: 'Auto: Generator 5', desc: 'Automatically purchases Generator 5 (when unlocked)', reqNode: 'Apex Autobuyers (auto_2)' },
    { key: 'gen6', name: 'Auto: Generator 6', desc: 'Automatically purchases Generator 6 (when unlocked)', reqNode: 'Apex Autobuyers (auto_2)' },
    { key: 'gen7', name: 'Auto: Generator 7', desc: 'Automatically purchases Generator 7 (when unlocked)', reqNode: 'Apex Autobuyers (auto_2)' },
    { key: 'gen8', name: 'Auto: Generator 8', desc: 'Automatically purchases Generator 8 (when unlocked)', reqNode: 'Apex Autobuyers (auto_2)' },
    { key: 'prestige', name: 'Prestige Autobuyer', desc: 'Automatically triggers Prestige using the selected reset condition', reqNode: 'Prestige Autobuyer (auto_3)' },

        <div>
          <div className="flex items-center gap-2 mb-1.5">
            <span className="px-2.5 py-0.5 rounded-full text-[10px] font-mono font-semibold uppercase tracking-wider bg-emerald-500/20 text-emerald-300 border border-emerald-500/30">

          <p className="text-xs text-slate-400 mt-1 max-w-xl leading-relaxed">

            <>
              <Check className="w-3.5 h-3.5 text-emerald-400" />
              <span className="text-emerald-400 font-bold">Saved!</span>
            </>
          ) : (
            <>
              <Save className="w-3.5 h-3.5 text-slate-400" />

/**
 * Robust Save System with automated backup rotation.
 * Rotates latest backup to prev_backup, and saves current state to primary and latest backup.
 */
export function saveStateWithBackup(state: GameState): boolean {
  if (typeof window === 'undefined') return false;
  try {
    const serialized = serializeState(state);
    const existing = localStorage.getItem(STORAGE_KEY);
    if (existing) {
      localStorage.setItem(STORAGE_PREV_BACKUP_KEY, existing);
    }
    localStorage.setItem(STORAGE_KEY, serialized);
    localStorage.setItem(STORAGE_BACKUP_KEY, serialized);
    return true;
  } catch (err) {
    console.error('Save failed:', err);
    return false;
  }
}

/**
 * Attempts to load from primary storage, falling back to backup if corrupted.
 */
export function loadStateSafe(): GameState {
  if (typeof window === 'undefined') return createInitialState();
  try {
    const primary = localStorage.getItem(STORAGE_KEY);
    if (primary) {
      const parsed = deserializeState(primary);
      if (parsed) return parsed;
    }

    const backup = localStorage.getItem(STORAGE_BACKUP_KEY);
    if (backup) {
      const parsed = deserializeState(backup);
      if (parsed) {
        console.warn('Recovered from latest backup save!');
        return parsed;
      }
    }

    const prevBackup = localStorage.getItem(STORAGE_PREV_BACKUP_KEY);
    if (prevBackup) {
      const parsed = deserializeState(prevBackup);
      if (parsed) {
        console.warn('Recovered from previous backup save!');
        return parsed;
      }
    }
  } catch (err) {
    console.error('Safe load failed:', err);
  }
  return createInitialState();
}

          // Auto Generators (only unlocked tiers)
          for (let i = 0; i < unlockedCount; i++) {
            const abKey = `gen${i + 1}`;
            const ab = state.autobuyers[abKey];
            if (ab && ab.unlocked && ab.enabled) {
              const gen = state.generators[i];
              if (state.infinityTree?.auto_4) {
                // Quantum Autobuyer: Buy max available
                while (true) {
                  const cost = getGeneratorCost(gen.baseCost, gen.costMult, gen.bought);
                  if (state.points.lt(cost)) break;
                  state.points = state.points.sub(cost);
                  gen.amount = gen.amount.add(1);
                  gen.bought++;
                }
              } else {
                const cost = getGeneratorCost(gen.baseCost, gen.costMult, gen.bought);
                if (state.points.gte(cost)) {
                  state.points = state.points.sub(cost);
                  gen.amount = gen.amount.add(1);
                  gen.bought++;
                }
              }
            }
          }

          // Auto Tickspeed
          const autoTick = state.autobuyers['tickspeed'];
          if (autoTick && autoTick.unlocked && autoTick.enabled) {
            const baseTick = getTickspeedBaseCost(state);
            if (state.infinityTree?.auto_4) {
              while (state.points.gte(state.tickspeed.cost)) {
                state.points = state.points.sub(state.tickspeed.cost);
                state.tickspeed.bought++;
                state.tickspeed.cost = baseTick.mul(state.tickspeed.costMult.pow(state.tickspeed.bought));
              }
            } else {
              const tickCost = state.tickspeed.cost;
              if (state.points.gte(tickCost)) {
                state.points = state.points.sub(tickCost);
                state.tickspeed.bought++;
                state.tickspeed.cost = getTickspeedCost(state);
              }
            }
          }

          // Auto Prestige (Branch 2 Node 3)
          const autoPres = state.autobuyers['prestige'];
          if (autoPres && autoPres.unlocked && autoPres.enabled) {
            const ppGain = calculatePrestigePointsGain(state.points, state);
            const target = typeof autoPres.target === 'number' ? autoPres.target : 1;
            const mode = autoPres.mode || 'targetPP'; // Keeps existing saves on their previous behavior.
            const previousGain = autoPres.previousPPGain instanceof Decimal
              ? autoPres.previousPPGain
              : Decimal.zero();
            const timeTarget = Math.max(1, autoPres.timeTarget || 60);
            const ratioTarget = Math.max(1.01, autoPres.ratioTarget || 2);
            let shouldPrestige = false;

            // Never reset until this run can actually award at least 1 PP.
            if (ppGain.gte(1)) {
              if (mode === 'time') {
                shouldPrestige = state.stats.timeThisPrestige >= timeTarget;
              } else if (mode === 'ppMultiplier') {
                // If there is no recorded previous run yet, use the first successful run
                // to establish a baseline; later runs must beat it by the configured ratio.
                shouldPrestige = previousGain.gt(0)
                  ? ppGain.gte(previousGain.mul(ratioTarget))
                  : true;
              } else {
                shouldPrestige = ppGain.gte(target);
              }
            }

            if (shouldPrestige) {
              const totalPP = state.prestige.points.add(ppGain);
              const completedPrestigeTime = state.stats.timeThisPrestige;
              // Trigger automated Prestige and remember the gain for ratio mode.
              state.prestige.count++;
              state.prestige.points = totalPP;
              state.stats.totalPrestiges++;
              state.stats.fastestPrestige = Math.min(state.stats.fastestPrestige, completedPrestigeTime);
              state.stats.highestPPEver = totalPP.gt(state.stats.highestPPEver) ? totalPP : state.stats.highestPPEver;
              state.autobuyers['prestige'] = { ...autoPres, previousPPGain: ppGain };
              state.stats.timeThisPrestige = 0;
              state.points = state.infinityTree?.pre_2 ? new Decimal(1e6) : (state.prestigeUpgrades?.tickspeedDiscount ? new Decimal(110) : new Decimal(10));
              state.generators = INITIAL_GENERATORS();
              if (state.infinityTree?.pre_2) {
                for (let k = 0; k < 4; k++) {
                  state.generators[k].amount = new Decimal(1);
                  state.generators[k].bought = 1;
                }
              }
              state.tickspeed.bought = 0;
              state.tickspeed.cost = getTickspeedBaseCost(state);
            }
          }
        }

        // Achievements check & Toast Notifications
        const { updated, newAchievements } = checkAchievements(state);
        state.achievements = updated;
        if (newAchievements.length > 0) {
          sound.playAchievement();
          newAchievements.forEach((ach) => {
            addToast({
              title: ach.name,
              message: ach.name,
              description: ach.desc,
              icon: '🏆',
              type: 'achievement',
            });
          });
        }

        // Auto Save to localStorage (if enabled in settings) with automated backup rotation
        if (state.settings.autoSaveEnabled) {
          saveAccumulator += dt;
          if (saveAccumulator >= state.settings.autoSaveInterval) {
            saveAccumulator = 0;
            state.lastUpdate = Date.now();
            saveStateWithBackup(state);
          }
        }

        return state;
      });
    };

    // When worker sends tick: if window is hidden / minimized, process background simulation
    if (worker) {
      worker.onmessage = () => {
        if (document.hidden) {
          const now = performance.now();
          const rawDt = (now - lastTime) / 1000;
          lastTime = now;
          if (rawDt > 0.01) {
            const maxStep = 0.25;
            const steps = Math.min(60, Math.ceil(rawDt / maxStep));
            const stepDt = rawDt / steps;
            for (let s = 0; s < steps; s++) {
              runSimulationStep(stepDt);
            }
          }
        }
      };
    }

    // Secondary interval heartbeat to guarantee active execution even if browser throttles workers
    const bgInterval = setInterval(() => {
      if (document.hidden) {
        const now = performance.now();
        const rawDt = (now - lastTime) / 1000;
        lastTime = now;
        if (rawDt > 0.05) {
          const maxStep = 0.25;
          const steps = Math.min(60, Math.ceil(rawDt / maxStep));
          const stepDt = rawDt / steps;
          for (let s = 0; s < steps; s++) {
            runSimulationStep(stepDt);
          }
        }
      }
    }, 100);

    // Main 60fps loop for active window
    const tick = (currentTime: number) => {
      if (!document.hidden) {
        const rawDt = (currentTime - lastTime) / 1000;
        lastTime = currentTime;
        const dt = Math.min(0.5, Math.max(0.001, rawDt));
        runSimulationStep(dt);
      }
      animId = requestAnimationFrame(tick);
    };

    // Immediate catch-up listener when window becomes visible or gains focus
    const handleFocusOrVisible = () => {
      const now = performance.now();
      const rawDt = (now - lastTime) / 1000;
      lastTime = now;
      if (rawDt > 0.05) {
        // Step in increments of at most 0.25s up to 100 steps so cascading & autobuyers catch up cleanly
        const steps = Math.min(100, Math.ceil(rawDt / 0.25));
        const stepDt = rawDt / steps;
        for (let s = 0; s < steps; s++) {
          runSimulationStep(stepDt);
        }
      }
    };

    document.addEventListener('visibilitychange', handleFocusOrVisible);
    window.addEventListener('focus', handleFocusOrVisible);

    animId = requestAnimationFrame(tick);

    return () => {
      cancelAnimationFrame(animId);
      clearInterval(bgInterval);
      if (worker) {
        worker.postMessage('stop');
        worker.terminate();
      }
      document.removeEventListener('visibilitychange', handleFocusOrVisible);
      window.removeEventListener('focus', handleFocusOrVisible);
    };
  }, [checkAchievements, addToast]);

  // Actions
  const buyGenerator = useCallback((id: number) => {
    setGameState((prev) => {
      const unlockedCount = getUnlockedGeneratorsCount(prev);
      if (id > unlockedCount) return prev;
      const genIndex = id - 1;
      const gen = prev.generators[genIndex];
      if (!gen) return prev;

      const cost = getGeneratorCost(gen.baseCost, gen.costMult, gen.bought);
      if (prev.points.lt(cost)) return prev;

      sound.playBuy();

      const newGenerators = prev.generators.map((g, idx) => {
        if (idx === genIndex) {
          return {
            ...g,
            amount: g.amount.add(1),
            bought: g.bought + 1,
          };
        }
        return g;
      });

      return {
        ...prev,
        points: prev.points.sub(cost),
        generators: newGenerators,
        stats: {
          ...prev.stats,
          totalManualClicks: prev.stats.totalManualClicks + 1,
        },
      };
    });
  }, []);

  const buyMaxGenerator = useCallback((id: number) => {
    setGameState((prev) => {
      const unlockedCount = getUnlockedGeneratorsCount(prev);
      if (id > unlockedCount) return prev;
      const genIndex = id - 1;
      const gen = prev.generators[genIndex];
      if (!gen) return prev;

      let currentPoints = prev.points;
      let currentAmount = gen.amount;
      let currentBought = gen.bought;
      let purchases = 0;

      while (true) {
        const cost = getGeneratorCost(gen.baseCost, gen.costMult, currentBought);
        if (currentPoints.lt(cost) || purchases >= 50) break;
        currentPoints = currentPoints.sub(cost);
        currentAmount = currentAmount.add(1);
        currentBought++;
        purchases++;
      }

      if (purchases === 0) return prev;
      sound.playBuy();

      const newGenerators = prev.generators.map((g, idx) => {
        if (idx === genIndex) {
          return { ...g, amount: currentAmount, bought: currentBought };
        }
        return g;
      });

      return {
        ...prev,
        points: currentPoints,
        generators: newGenerators,
      };
    });
  }, []);

  const buyTickspeed = useCallback(() => {
    setGameState((prev) => {
      const cost = prev.tickspeed.cost;
      if (prev.points.lt(cost)) return prev;

      sound.playBuy();
      const newBought = prev.tickspeed.bought + 1;
      const nextCost = getTickspeedBaseCost(prev).mul(prev.tickspeed.costMult.pow(newBought));

      return {
        ...prev,
        points: prev.points.sub(cost),
        tickspeed: {
          ...prev.tickspeed,
          bought: newBought,
          cost: nextCost,

        stats: {
          ...prev.stats,
          totalManualClicks: prev.stats.totalManualClicks + 1,
        },
      };
    });
  }, []);

  const buyMaxAll = useCallback(() => {
    setGameState((prev) => {
      let currentPoints = prev.points;
      const newGenerators = prev.generators.map((g) => ({ ...g }));
      const unlockedCount = getUnlockedGeneratorsCount(prev);
      let didBuy = false;

      // Buy Tickspeed
      let tickBought = prev.tickspeed.bought;
      let tickCost = prev.tickspeed.cost;
      const baseTick = getTickspeedBaseCost(prev);
      while (currentPoints.gte(tickCost)) {
        currentPoints = currentPoints.sub(tickCost);
        tickBought++;
        tickCost = baseTick.mul(prev.tickspeed.costMult.pow(tickBought));
        didBuy = true;
      }

      // Buy all unlocked generators
      for (let i = unlockedCount - 1; i >= 0; i--) {
        const g = newGenerators[i];
        while (true) {
          const cost = getGeneratorCost(g.baseCost, g.costMult, g.bought);
          if (currentPoints.lt(cost)) break;
          currentPoints = currentPoints.sub(cost);
          g.amount = g.amount.add(1);
          g.bought++;
          didBuy = true;
        }
      }

      if (!didBuy) return prev;
      sound.playBuy();

      return {
        ...prev,
        points: currentPoints,
        generators: newGenerators,
        tickspeed: {
          ...prev.tickspeed,
          bought: tickBought,
          cost: tickCost,
        },
      };
    });
  }, []);

  const performPrestige = useCallback(() => {
    setGameState((prev) => {
      const ppGain = calculatePrestigePointsGain(prev.points, prev);
      if (ppGain.lt(1)) return prev;

      sound.playCollapse();

      const totalPrestiges = prev.stats.totalPrestiges + 1;
      const fastest = Math.min(prev.stats.fastestPrestige, prev.stats.timeThisPrestige);
      const resetGenerators = INITIAL_GENERATORS();
      const newPP = prev.prestige.points.add(ppGain);

      const baseTickCost = prev.prestigeUpgrades.tickspeedDiscount ? new Decimal(10) : new Decimal(1000);
      let startingPoints = prev.prestigeUpgrades.tickspeedDiscount ? new Decimal(110) : new Decimal(10);
      if (prev.infinityTree?.pre_2) {
        startingPoints = new Decimal(1e6);
        for (let i = 0; i < 4; i++) {
          resetGenerators[i].amount = new Decimal(1);
          resetGenerators[i].bought = 1;
        }
      }

      return {
        ...prev,
        points: startingPoints,
        prestige: {
          ...prev.prestige,
          count: prev.prestige.count + 1,
          points: newPP,
        },
        tickspeed: {
          ...prev.tickspeed,
          bought: 0,
          cost: baseTickCost,
        },
        generators: resetGenerators,
        autobuyers: {
          ...prev.autobuyers,
          prestige: prev.autobuyers.prestige
            ? { ...prev.autobuyers.prestige, previousPPGain: ppGain }
            : prev.autobuyers.prestige,
        },
        stats: {
          ...prev.stats,
          totalPrestiges,
          fastestPrestige: fastest,
          timeThisPrestige: 0,
          highestPPEver: newPP.gt(prev.stats.highestPPEver) ? newPP : prev.stats.highestPPEver,
        },
      };
    });
  }, []);

  // Mini-reset: Prestige Energy
  // Resets everything prior to Prestige Energy (including prestige points, prestige count, and ALL prestige upgrades)
  // in exchange for a permanent compounded Generators multiplier!
  const performPrestigeEnergy = useCallback(() => {
    setGameState((prev) => {
      if (prev.points.lt(PRESTIGE_ENERGY_THRESHOLD)) return prev;

      const gainedMult = calculatePrestigeEnergyMultiplier(prev.points);
      if (gainedMult.lt(2)) return prev;

      sound.playBoost();

      const newMultiplier = prev.prestigeEnergy.multiplier.mul(gainedMult);
      const newAmount = prev.prestigeEnergy.amount.add(gainedMult);
      const initial = createInitialState();

      return {
        ...prev,
        points: new Decimal(10),
        generators: INITIAL_GENERATORS(),
        tickspeed: {
          ...initial.tickspeed,
        },
        prestige: {
          count: 0,
          points: Decimal.zero(),
          unlockedGenerators: 4,
        },
        prestigeUpgrades: {
          ...initial.prestigeUpgrades,
        },
        prestigeEnergy: {
          unlocked: true,
          amount: newAmount,
          multiplier: newMultiplier,
          resets: prev.prestigeEnergy.resets + 1,
        },
        stats: {
          ...prev.stats,
          timeThisPrestige: 0,
        },
      };
    });
  }, []);

  // Buy any Prestige Upgrade
  const buyPrestigeUpgrade = useCallback((upgradeId: string) => {
    setGameState((prev) => {
      const currentPP = prev.prestige.points;
      const upgs = prev.prestigeUpgrades;

      const costs: Record<string, Decimal> = {
        pointScaling: new Decimal(1),
        achievementBoost: new Decimal(2),
        tickspeedDiscount: new Decimal(2),
        prestigeSynergy: new Decimal(5),
        amplifiedDoubling: new Decimal(15),
        chronosFlow: new Decimal(25),
        logarithmicCascade: new Decimal(50),
        cascadeSurge: new Decimal(100),
        cosmicResonance: new Decimal(250),
      };

      // Upgrade 3A: Cascading Boosts (up to 20 levels)
      if (upgradeId === 'cascadingBoosts') {
        const currentLvl = upgs.cascadingBoostsLevel;
        if (currentLvl >= MAX_CASCADING_BOOSTS) return prev;
        const cost = getCascadingBoostCost(currentLvl);
        if (currentPP.lt(cost)) return prev;
        sound.playBuy();

        return {
          ...prev,
          prestige: {
            ...prev.prestige,
            points: currentPP.sub(cost),
          },
          prestigeUpgrades: {
            ...upgs,
            cascadingBoostsLevel: currentLvl + 1,
          },
        };
      }

      // Upgrade 3B: Generator Expander (up to 4 levels, unlocking Gen 5, 6, 7, 8)
      if (upgradeId === 'generatorUnlock') {
        const currentLvl = upgs.generatorUnlockLevel;
        if (currentLvl >= MAX_GENERATOR_UNLOCKS) return prev;
        const cost = getGeneratorUnlockCost(currentLvl);
        if (currentPP.lt(cost)) return prev;
        sound.playBuy();

        const nextLvl = currentLvl + 1;
        const nextUnlockedGens = 4 + nextLvl;

        return {
          ...prev,
          prestige: {
            ...prev.prestige,
            points: currentPP.sub(cost),
            unlockedGenerators: nextUnlockedGens,
          },
          prestigeUpgrades: {
            ...upgs,
            generatorUnlockLevel: nextLvl,
          },
        };
      }

      const cost = costs[upgradeId];
      if (!cost || currentPP.lt(cost)) return prev;

      // Check if already bought boolean upgrade
      if ((upgs as unknown as Record<string, boolean>)[upgradeId]) return prev;

      sound.playBuy();

      const nextUpgrades = {
        ...upgs,
        [upgradeId]: true,
      };

      const nextState = {
        ...prev,
        prestige: { ...prev.prestige, points: currentPP.sub(cost) },
        prestigeUpgrades: nextUpgrades,
      };

      if (upgradeId === 'tickspeedDiscount') {
        return {
          ...nextState,
          points: nextState.points.add(100),
          tickspeed: {
            ...nextState.tickspeed,
            cost: getTickspeedCost(nextState),
          },
        };
      }

      return nextState;
    });
  }, []);

  // Complete reset to Infinity layer: grants IP, unlocks Infinity tab, and resets pre-infinity progress
  const proceedInfinity = useCallback(() => {
    setShowInfinityOverlay(false);
    infinityTriggeredRef.current = false;

    setGameState((prev) => {

      let ipGain = 1;
      if (prev.infinityTree?.feat_4 && prev.points.gte(INFINITY_THRESHOLD)) {
        const excess = Math.max(0, prev.points.log10() - 308);
        ipGain += Math.floor(excess / 10);
      }
      const newIP = prev.infinity.points.add(ipGain);

      let startingPoints = new Decimal(10);
      if (prev.infinityTree?.pre_2) {
        startingPoints = new Decimal(1e6);
        for (let i = 0; i < 4; i++) {

          resetGenerators[i].bought = 1;

        ...(prev.infinityTree || {}),
        root: true,

        points: startingPoints,
        generators: resetGenerators,
        tickspeed: {

          bought: 0,

        prestige: {

  proceedInfinityRef.current = proceedInfinity;

  const buyInfinityTreeNode = useCallback((nodeId: string) => {

          if (nextAutobuyers[k]) nextAutobuyers[k] = { ...nextAutobuyers[k], unlocked: true };
        });
      } else if (nodeId === 'auto_3') {
        if (nextAutobuyers['prestige']) nextAutobuyers['prestige'] = { ...nextAutobuyers['prestige'], unlocked: true };

      const nextState: GameState = {
        ...prev,
        infinity: {
          ...prev.infinity,
          points: nextIP,

        infinityTree: nextTree,
        autobuyers: nextAutobuyers,

      return nextState;

  }, []);

  const exportSave = useCallback((): string => {

  const importSave = useCallback((raw: string): boolean => {

      let json = raw.trim();
      try {
        json = decodeURIComponent(escape(atob(json)));

  const saveGame = useCallback(() => {
    setGameState((prev) => {
      const now = Date.now();
      const updated = {

        lastUpdate: now,
      };
      const saved = saveStateWithBackup(updated);
      if (saved) {

import { Decimal } from './decimal';
import { GameState } from '../types/game';

// Prestige unlock threshold (1.00e18 Points)
export const PRESTIGE_REQUIREMENT = new Decimal(1e18);

export function getPrestigeRequirement(_state?: GameState): Decimal {
  return PRESTIGE_REQUIREMENT;
}

// Prestige Energy unlock threshold (1.00e154 Points)
export const PRESTIGE_ENERGY_THRESHOLD = new Decimal(1, 154);

// Infinity threshold (1.79e308 Points)
export const INFINITY_THRESHOLD = new Decimal(1.7976931348623157, 308);

/**
 * Returns number of unlocked generators:
 * Initially 4, unlocked sequentially up to 8 via Generator Expander (Levels 1 to 4)
 */
export function getUnlockedGeneratorsCount(state: GameState): number {
  const unlockLvl = state.prestigeUpgrades?.generatorUnlockLevel || 0;
  return Math.min(8, Math.max(4, 4 + unlockLvl));
}

/**
 * Returns the cost for the next purchase of generator
 */
export function getGeneratorCost(baseCost: Decimal, costMult: Decimal, bought: number): Decimal {
  return baseCost.mul(costMult.pow(bought));
}

/**
 * Upgrade 3A: Cascading Boosts
 * Like Dimensional Boosts in Antimatter Dimensions (without resetting!):
 * Level 1: doubles Generator 1.
 * Level 2: doubles Generator 1 again, and doubles Generator 2.
 * Level 3: doubles Generator 1 again, Generator 2 again, and Generator 3.
 * Level L: For any generator i (1..8), if L >= i, it gets 2^(L - i + 1).
 * After level 8 (levels 9..20), each additional level doubles ALL 8 generators!
 */
export const MAX_CASCADING_BOOSTS = 20;

export function getCascadingBoostCost(currentLevel: number): Decimal {
  if (currentLevel <= 8) {
    return new Decimal(4).pow(currentLevel);
  }
  const base8 = new Decimal(4).pow(8);
  return base8.mul(new Decimal(8).pow(currentLevel - 8));
}

export function getCascadingBoostMultiplierForGen(genId: number, boostLevel: number): Decimal {
  if (boostLevel < genId) return Decimal.one();
  return new Decimal(2).pow(boostLevel - genId + 1);
}

/**
 * Upgrade 3B: Generator Expander (Unlocks Generators 5, 6, 7, 8)
 * Level 1 (Cost 50 PP): Unlocks Generator 5
 * Level 2 (Cost 750 PP): Unlocks Generator 6
 * Level 3 (Cost 7,500 PP): Unlocks Generator 7
 * Level 4 (Cost 125,000 PP): Unlocks Generator 8
 */
export const MAX_GENERATOR_UNLOCKS = 4;
export const GENERATOR_UNLOCK_COSTS: Decimal[] = [
  new Decimal(50),
  new Decimal(750),
  new Decimal(7500),
  new Decimal(125000),
];

export function getGeneratorUnlockCost(currentLevel: number): Decimal {
  if (currentLevel >= MAX_GENERATOR_UNLOCKS) return new Decimal(Infinity);
  return GENERATOR_UNLOCK_COSTS[currentLevel];
}

/**
 * Multiplier for generator based on purchases, Upgrade 3A (Cascading Boosts), Upgrade 9, Prestige Energy,
 * and the Infinity Tree Root Node (4^Infinities) + tree branches
 */
export function getGeneratorMultiplier(genId: number, state: GameState): Decimal {
  const gen = state.generators[genId - 1];
  if (!gen) return Decimal.one();

  // Base purchase doubling: 2x, or 2.25x with Upgrade 6 (Amplified Doubling)
  const baseDoubling = state.prestigeUpgrades?.amplifiedDoubling ? 2.25 : 2.0;
  let mult = new Decimal(baseDoubling).pow(gen.bought);

  // Upgrade 3A: Cascading Boosts (AD-style non-resetting dimensional boosts)
  const boostLevel = state.prestigeUpgrades?.cascadingBoostsLevel || 0;
  if (boostLevel >= genId) {
    const boostMult = getCascadingBoostMultiplierForGen(genId, boostLevel);
    mult = mult.mul(boostMult);
  }

  // Upgrade 9: Cascade Surge (Higher tiers 5-8 produce 10x faster)
  if (genId >= 5 && state.prestigeUpgrades?.cascadeSurge) {
    mult = mult.mul(10);
  }

  // Upgrade 10: Cosmic Resonance (Every 10 total generator purchases multiplies each generator by 1.1x)
  if (state.prestigeUpgrades?.cosmicResonance) {
    const totalPurchases = state.generators.reduce((sum, g) => sum + g.bought, 0);
    const groupsOf10 = Math.floor(totalPurchases / 10);
    if (groupsOf10 > 0) {
      mult = mult.mul(new Decimal(1.1).pow(groupsOf10));
    }
  }

  // Prestige Energy Generators Multiplier (applies to ALL generators)
  if (state.prestigeEnergy?.multiplier && state.prestigeEnergy.multiplier.gt(1)) {
    mult = mult.mul(state.prestigeEnergy.multiplier);
  }

  // Infinity Tree - Root Node: 4^Infinities to all generators
  const totalInfinities = state.infinity?.count || state.stats?.totalInfinities || 0;
  if (totalInfinities > 0) {
    mult = mult.mul(new Decimal(4).pow(totalInfinities));
  }

  // Infinity Tree - Temporal Surge: starts at x1.00 and adds +0.01x per second in the current Infinity.
  if (state.infinityTree?.pre_1) {
    const timeInf = state.stats?.timeThisInfinity || 0;
    const timeMult = 1 + (timeInf * 0.01);
    mult = mult.mul(new Decimal(timeMult));
  }

  // Infinity Tree - Pre-Infinity Node 4 (Tickspeed Induction: tickspeed affects generators at reduced rate)
  if (state.infinityTree?.pre_4) {
    const tickspeedSpeed = getTotalTickspeedSpeed(state);
    if (tickspeedSpeed.gt(1)) {
      mult = mult.mul(tickspeedSpeed.pow(0.5));
    }
  }

  return mult;
}

/**
 * Tickspeed base cost:
 * Upgrade 4 reduces initial base cost by 100x (from 1,000 down to 10!)
 */
export function getTickspeedBaseCost(state: GameState): Decimal {
  return state.prestigeUpgrades?.tickspeedDiscount ? new Decimal(10) : new Decimal(1000);
}

/**
 * Current tickspeed cost based on purchases
 */
export function getTickspeedCost(state: GameState): Decimal {
  const base = getTickspeedBaseCost(state);
  return base.mul(state.tickspeed.costMult.pow(state.tickspeed.bought));
}

/**
 * Current tickspeed rate multiplier:
 * Base is 1.10x, upgraded to 1.15x with Upgrade 7 (Chronos Flow)
 */
export function getTotalTickspeedSpeed(state: GameState): Decimal {
  const basePower = state.prestigeUpgrades?.chronosFlow ? 1.15 : state.tickspeed.multiplier;
  return new Decimal(basePower).pow(state.tickspeed.bought);
}

/**
 * Points generation multiplier derived from Prestige Upgrades
 */
export function getPointsGenerationMultiplier(state: GameState): Decimal {
  let mult = Decimal.one();

  // Upgrade 1: 2x per 1e6 (6 OoM) Points
  if (state.prestigeUpgrades?.pointScaling && state.points.gte(1e6)) {
    const oom6 = Math.floor(state.points.log10() / 6);
    if (oom6 > 0) {
      mult = mult.mul(new Decimal(2).pow(oom6));
    }
  }

  // Upgrade 2: 1.05x per achievement unlocked (multiplicative)
  if (state.prestigeUpgrades?.achievementBoost) {
    const unlockedCount = Object.values(state.achievements).filter(Boolean).length;
    if (unlockedCount > 0) {
      mult = mult.mul(new Decimal(1.05).pow(unlockedCount));
    }
  }

  // Upgrade 5: Prestige Synergy (Points boosted by unspent PP without exponent: 1 + PP, hardcapped at 1e10)
  if (state.prestigeUpgrades?.prestigeSynergy && state.prestige.points.gt(0)) {
    const rawPpBoost = state.prestige.points.add(1);
    const ppBoost = rawPpBoost.gt(1e10) ? new Decimal(1e10) : rawPpBoost;
    mult = mult.mul(ppBoost);
  }

  // Upgrade 8: Logarithmic Cascade (Points boosted by log10(points)^2)
  if (state.prestigeUpgrades?.logarithmicCascade && state.points.gte(10)) {
    const logP = state.points.log10();
    mult = mult.mul(new Decimal(logP * logP));
  }

  return mult;
}

/**
 * Calculate Prestige Points gained on Prestige reset.
 * Requirement: 1.00e18 Points.
 * Buffed curve: 1 PP at requirement, scaling so 1e200 gives about 1.9e7 PP.
 * Pre-Infinity Node 3 multiplies PP gain by 3x.
 */
export function calculatePrestigePointsGain(points: Decimal, state?: GameState): Decimal {
  if (points.lt(PRESTIGE_REQUIREMENT)) {
    return Decimal.zero();
  }

  const logPoints = points.log10();
  const reqLog = PRESTIGE_REQUIREMENT.log10();
  const excessLog = Math.max(0, logPoints - reqLog);
  // Use Decimal exponentiation instead of Math.pow so later layers can safely exceed 1e308.
  let gain = new Decimal(10).pow(excessLog / 25).floor();
  if (gain.lt(1)) gain = new Decimal(1);

  if (state?.infinityTree?.pre_3) {
    gain = gain.mul(3);
  }

  return gain;
}

/**
 * Calculate Prestige Energy Generators multiplier gained on Prestige Energy reset:
 * Unlocks at 1e154 Points.
 * Weakened multiplier curve: base x1.50, scaling gently with (points / 1e154)^0.016
 */
export function calculatePrestigeEnergyMultiplier(points: Decimal): Decimal {
  if (points.lt(PRESTIGE_ENERGY_THRESHOLD)) {
    return Decimal.zero();
  }

  const excessLog = Math.max(0, points.log10() - 154);
  const scaledFactor = Math.pow(10, excessLog / 60);
  const totalMult = 1.5 * Math.max(1, scaledFactor);

  return new Decimal(totalMult);
}
